>>> text_indentation = __import__('5-text_indentation').text_indentation

# Basic behavior with separators
>>> text_indentation("Hello. World? Yes: indeed.")
Hello.
<BLANKLINE>
World?
<BLANKLINE>
Yes:
<BLANKLINE>
indeed.

# Leading/trailing spaces are trimmed per printed line
>>> text_indentation("  A  .   B   ?   C   :   D   ")
A.
<BLANKLINE>
B?
<BLANKLINE>
C:
<BLANKLINE>
D

# No separators: prints once, trimmed
>>> text_indentation("   No separators here   ")
No separators here

# Multiple spaces after separators are skipped
>>> text_indentation("End.   Next?    Then:   Final")
End.
<BLANKLINE>
Next?
<BLANKLINE>
Then:
<BLANKLINE>
Final

# Mixed content without trailing separator
>>> text_indentation("Alpha beta: gamma delta")
Alpha beta:
<BLANKLINE>
gamma delta
